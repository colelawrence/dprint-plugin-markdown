== should format complex top-level jsx only through safe embedded output ==
<Foo format_mdx_jsx_complex='yes'>
      <Bar    >hi    </Bar>
       {  hello       }
       {     /* another comment */}
</Foo>

[expect]
<Foo format_mdx_jsx_complex="yes">
  <Bar>hi</Bar>
  {hello}
  {/* another comment */}
</Foo>
